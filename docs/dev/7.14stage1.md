3. 已实现的核心后端源码规范
3.1 核心底层数据引擎：core/database.py
采用 SQLite 作为本地轻量级、跨平台数据存储。设计了两张核心表，具备天然的去重逻辑（UNIQUE 约束）与级联删除（CASCADE）。  

        import sqlite3  
        from datetime import datetime  
        class DBManager:  
            def __init__(self, db_path="database.db"):  
                self.conn = sqlite3.connect(db_path)  
                self.create_tables()  

            def create_tables(self):
                with self.conn:
                    # 1. 订阅源表
                    self.conn.execute("""
                        CREATE TABLE IF NOT EXISTS feeds (
                            id INTEGER PRIMARY KEY AUTOINCREMENT,
                            title TEXT,
                            xml_url TEXT UNIQUE,
                            html_url TEXT
                        )
                    """)
                    # 2. 文章表（使用 link 作为唯一约束，实现去重）
                    self.conn.execute("""
                        CREATE TABLE IF NOT EXISTS articles (
                            id INTEGER PRIMARY KEY AUTOINCREMENT,
                            feed_id INTEGER,
                            title TEXT,
                            link TEXT UNIQUE,
                            description TEXT,
                            published TEXT,
                            FOREIGN KEY(feed_id) REFERENCES feeds(id) ON DELETE CASCADE
                        )
                    """)

            def add_feed(self, title, xml_url, html_url=""):
                try:
                    with self.conn:
                        cursor = self.conn.execute(
                            "INSERT INTO feeds (title, xml_url, html_url) VALUES (?, ?, ?)",
                            (title, xml_url, html_url)
                        )
                        return cursor.lastrowid
                except sqlite3.IntegrityError:
                    cursor = self.conn.execute("SELECT id FROM feeds WHERE xml_url = ?", (xml_url,))
                    return cursor.fetchone()[0]

            def get_all_feeds(self):
                cursor = self.conn.execute("SELECT id, title, xml_url FROM feeds")
                return cursor.fetchall()

            def save_articles(self, feed_id, entries):
                with self.conn:
                    for entry in entries:
                        title = entry.get("title", "No Title")
                        link = entry.get("link", "")
                        description = entry.get("summary", entry.get("description", ""))
                        published = entry.get("published", str(datetime.now()))
                        try:
                            self.conn.execute(
                                "INSERT INTO articles (feed_id, title, link, description, published) VALUES (?, ?, ?, ?, ?)",
                                (feed_id, title, link, description, published)
                            )
                        except sqlite3.IntegrityError:
                            continue  # 重复文章自动跳过

            def get_articles_by_feed(self, feed_id):
                cursor = self.conn.execute(
                    "SELECT id, title, published FROM articles WHERE feed_id = ? ORDER BY id DESC", (feed_id,)
                )
                return cursor.fetchall()

            def get_article_detail(self, article_id):
                cursor = self.conn.execute(
                    "SELECT title, description, link FROM articles WHERE id = ?", (article_id,)
                )
                return cursor.fetchone()

3.2 业务用例与文件解析：domain/feed/
domain/feed/opml_parser.py  

    import xml.etree.ElementTree as ET

    def parse_opml_file(file_path: str) -> list[dict]:
        """解析标准 OPML 文件，提取订阅源列表"""
        feeds = []
        tree = ET.parse(file_path)
        root = tree.getroot()
        for outline in root.findall(".//outline"):
            xml_url = outline.get("xmlUrl")
            if xml_url:
                feeds.append({
                    "title": outline.get("text", outline.get("title", "未命名源")),
                    "xml_url": xml_url,
                    "html_url": outline.get("htmlUrl", "")
                })
        return feeds

domain/feed/use_cases.py  

    import feedparser  
    import requests  
    from core.database import DBManager  
    from domain.feed.opml_parser import parse_opml_file  

    class FeedUseCase:
        def __init__(self, db: DBManager):
            self.db = db

        def add_single_feed(self, xml_url: str) -> bool:
            """单源网络抓取与解析"""
            try:
                res = requests.get(xml_url, timeout=10)
                feed_data = feedparser.parse(res.text)
                if feed_data.bozo:
                    return False
                feed_title = feed_data.feed.get("title", xml_url)
                html_url = feed_data.feed.get("link", "")
                
                feed_id = self.db.add_feed(feed_title, xml_url, html_url)
                self.db.save_articles(feed_id, feed_data.entries)
                return True
            except Exception:
                return False

        def refresh_all(self):
            """全网遍历刷新"""
            feeds = self.db.get_all_feeds()
            for feed_id, title, xml_url in feeds:
                try:
                    res = requests.get(xml_url, timeout=10)
                    feed_data = feedparser.parse(res.text)
                    self.db.save_articles(feed_id, feed_data.entries)
                except Exception:
                    continue

        def import_from_opml(self, file_path: str) -> int:
            """批量导入 OPML 并入库"""
            parsed_feeds = parse_opml_file(file_path)
            count = 0
            for f in parsed_feeds:
                self.db.add_feed(f["title"], f["xml_url"], f["html_url"])
                count += 1
            return count

5. 接续开发指南：PySide6 前后端对接契约
由于全栈语言统一为 Python，且框架换成了 Qt (PySide6)，为了防止 UI 主线程在处理网络IO或大规模数据库读写时产生卡顿僵死（Not Responding），任何后续承接开发的 AI 或人类开发者必须遵循以下对接契约：  
5.1 异步线程协同设计 (ui/workers.py)
严禁在 UI 页面类中直接调用 FeedUseCase.refresh_all() 或 add_single_feed()。必须使用 QThread 配合 Signal 异步中转：
# ui/workers.py
    from PySide6.QtCore import QThread, Signal  
    from domain.feed.use_cases import FeedUseCase  

    class RefreshFeedsWorker(QThread):  
        """全网刷新异步 Worker"""  
        started = Signal()
        finished = Signal(bool, str)

        def __init__(self, feed_use_case: FeedUseCase):
            super().__init__()
            self.feed_use_case = feed_use_case

        def run(self):
            self.started.emit()
            try:
                self.feed_use_case.refresh_all()
                self.finished.emit(True, "所有订阅源刷新成功！")
            except Exception as e:
                self.finished.emit(False, str(e))

5.2 依赖注入启动入口 (main.py)
在程序的启动总入口中进行依赖注入，实例化后端核心，并作为上下文资源提供给 PySide6 窗口组件。
# main.py
    import sys
    from PySide6.QtWidgets import QApplication
    from core.database import DBManager
    from domain.feed.use_cases import FeedUseCase
    from ui.main_window import MainWindow

    def main():
        # 1. 实例化核心基础设施
        db = DBManager("database.db")
        feed_services = FeedUseCase(db)
    
        # 2. 启动 Qt 运行时
        app = QApplication(sys.argv)
        
        # 3. 依赖注入：将后端逻辑解耦传入主窗体
        window = MainWindow(feed_services)
        window.show()
        
        sys.exit(app.exec())

        if __name__ == "__main__":
            main()
